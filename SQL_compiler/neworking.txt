SELECT S.SURNAME, 'успевает'
FROM STUDENT S
WHERE 2 < ALL (SELECT MARK
FROM EXAM_MARKS EX
WHERE S.STUDENT_ID = EX.STUDENT_ID)
UNION
SELECT S.SURNAME, 'не успевает'
FROM STUDENT S
WHERE 2 = ANY(SELECT MARK
FROM EXAM_MARKS EX
WHERE S.STUDENT_ID = EX.STUDENT_ID)
UNION
SELECT S.SURNAME, 'не сдавал'
FROM STUDENT S
WHERE NOT EXISTS (SELECT MARK
FROM EXAM_MARKS EX
WHERE S.STUDENT_ID = EX.STUDENT_ID)
ORDER BY SURNAME; --any all exists, должно быть 2889
